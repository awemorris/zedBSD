[pi5]
kernel=vmunix
# vmunix is linked to run at 0x80000, just above the secure firmware (BL31)
# the bootloader keeps at 0..0x80000.
kernel_address=0x80000
# Keep the PCIe link to RP1 and its BARs as the firmware set them, so the
# kernel reaches RP1's USB and Ethernet without training the link again.
pciex4_reset=0
# Enable the external PCIe port (the FFC connector of the M.2 HATs), which
# the kernel resets and trains itself; Gen 2, as the device tree says.
# No device_tree line: the bootloader picks bcm2712-rpi-5-b.dtb (C1) or
# bcm2712d0-rpi-5-b.dtb (D0) by the chip's stepping, and both are here.
# The firmware framebuffer the console draws on: 32 bits per pixel, a
# 1920x1080 surface, no overscan border.
framebuffer_depth=32
framebuffer_ignore_alpha=1
framebuffer_width=1920
framebuffer_height=1080
disable_overscan=1
[all]
